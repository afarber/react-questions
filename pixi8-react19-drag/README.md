# Creating the pixi8-react19-drag app

    mkdir pixi8-react19-drag
    cd pixi8-react19-drag
    npm create vite@latest .
    npm install
    npm install react-router react-router-dom @react-hook/media-query @pixi/react pixi.js@^8.9.0
    npm run dev
